const API_BASE = 'http://localhost:8000'; // or your production URL

// Auth state
let isAuthenticated = false;

// === AUTHENTICATION ===
async function authenticate() {
  authStatus.textContent = 'Authenticating with backend...';
  authStatus.style.color = '#facc15';
  try {
    const res = await fetch(`${API_BASE}/api/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        email: 'rahmann@apexlifeglobal.com',
        password: 'your-secure-password' // in production, prompt for it
      })
    });
    if (!res.ok) throw new Error('Auth failed');
    const data = await res.json();
    authToken = data.token || 'dummy-token';
    localStorage.setItem('authToken', authToken);
    isAuthenticated = true;
    userLane = data.lane || 'owner';
    authStatus.textContent = `✅ Authenticated as ${userLane}`;
    authStatus.style.color = '#22c55e';
    authOverlay.style.display = 'none';
    vaultContainer.style.display = 'block';
    renderVault();
    fetchMetrics();
    startMetricPolling();
  } catch (err) {
    authStatus.textContent = '❌ Auth failed — check backend';
    authStatus.style.color = '#ef4444';
  }
}

authBtn.addEventListener('click', authenticate);

// Check session on load
if (authToken) {
  fetch(`${API_BASE}/api/auth/verify`, {
    headers: { 'Authorization': `Bearer ${authToken}` }
  })
  .then(res => {
    if (res.ok) {

}

    document.getElementById('sessions').textContent = data.sessions || '3';

    document.getElementById('sessions').textContent = Math.floor(1 + Math.random() * 5);
    document.getElementById('cache').textContent = Math.floor(80 + Math.random() * 15) + '%';

}

function auth(req, res, next) {
  const token = req.cookies["omni_vault_token"];
  if (!token) return res.status(401).json({ error: "unauthorized" });

  try {
    const payload = jwt.verify(token, JWT_SECRET);
    req.user = payload;
    next();
  } catch {
    return res.status(401).json({ error: "invalid_token" });
  }
}

function requireRole(role) {
  return (req, res, next) => {
    if (!req.user || (req.user.role !== role && req.user.role !== "owner")) {
      return res.status(403).json({ error: "forbidden" });
    }
    next();
  };
}

async function authenticate() {
  authStatus.textContent = 'Authenticating with backend...';
  authStatus.style.color = '#facc15';
  try {
    const res = await fetch(`${API_BASE}/api/auth/login`, {
      method: 'POST',

      body: JSON.stringify({
        email: 'rahmann@apexlifeglobal.com',
        password: 'your-secure-password' // in production, prompt for it
      })
    });
    if (!res.ok) throw new Error('Auth failed');
    const data = await res.json();
    authToken = data.token || 'dummy-token';
    localStorage.setItem('authToken', authToken);
    isAuthenticated = true;
    userLane = data.lane || 'owner';
    authStatus.textContent = `✅ Authenticated as ${userLane}`;

async function fetchMetrics() {
  if (!authToken) return;
  try {
    const res = await fetch(`${API_BASE}/api/system/metrics`, {
      headers: { 'Authorization': `Bearer ${authToken}` }
    });
    if (!res.ok) throw new Error('Metrics unavailable');
    const data = await res.json();
    document.getElementById('uptime').textContent = data.uptime || '99.99%';
    document.getElementById('requests').textContent = data.requests || '1,247';
    document.getElementById('sessions').textContent = data.sessions || '3';
    document.getElementById('cache').textContent = data.cache_hit || '87%';
  } catch {
    // fallback to simulated
    document.getElementById('uptime').textContent = '99.99%';
    document.getElementById('requests').textContent = Math.floor(1000 + Math.random() * 500);
    document.getElementById('sessions').textContent = Math.floor(1 + Math.random() * 5);
    document.getElementById('cache').textContent = Math.floor(80 + Math.random() * 15) + '%';
  }
}
